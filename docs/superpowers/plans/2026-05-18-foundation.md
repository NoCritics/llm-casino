# AI-Run Casino — Plan 1 of 8: Foundation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Stand up the monorepo substrate — pnpm + Turborepo, two skeleton Next.js apps (casino + dashboard), a Neon Postgres database with the V1 schema from the design spec, Vercel-linked deploys, and the `/org`, `/sim`, `/agents` skeleton directories. After this plan the casino has no games and no agents — but every later plan can build on a real foundation.

**Architecture:** Single git repo, pnpm + Turborepo workspaces. Two Next.js 16 (App Router, Fluid Compute) apps under `apps/`. One shared package `packages/db` owning Prisma schema + client. Neon Postgres provisioned via the Vercel Marketplace; `DATABASE_URL` populated by the Marketplace integration. Deployed via `vercel link` on each app — `git push` triggers preview/production deploys automatically.

**Tech Stack:** pnpm, Turborepo, Next.js 16 (App Router), Node.js 24, TypeScript, Prisma, Neon Postgres (Vercel Marketplace), Vitest, GitHub Actions, Vercel.

**Spec reference:** `docs/superpowers/specs/2026-05-18-ai-run-casino-design.md` — Sections 6 (state storage), 13 (tech stack), Appendix A (repo layout).

## Conventions used in this plan

- All paths are relative to repo root unless prefixed with `apps/<app>/...` etc.
- Commands assume the engineer is at repo root (`cd C:\Users\user\source\repos\workstation\casino`).
- The host is Windows + PowerShell, but commands shown are cross-shell unless noted. Where PowerShell differs (no `&&`, env var syntax) the PS form is given.
- `pnpm` is the package manager. **Do not use npm or yarn** anywhere in this monorepo — mixing lockfiles will break Turborepo caching.
- The repo is already initialized as a git repo with one commit (the design spec).

## Task 9: Author the V1 Prisma schema

**Files:**
- Modify: `packages/db/prisma/schema.prisma`

- [ ] **Step 1: Replace the Prisma schema with the full V1 model**

`packages/db/prisma/schema.prisma`:

```prisma
generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

// --- Product state ---

enum PersonaKind {
  whale
  grinder
  casual
  refund_seeker
  bonus_abuser
  lurker
  human_observer
}

model Player {
  id           String       @id @default(cuid())
  personaKind  PersonaKind
  balance      BigInt       @default(0)        // virtual chips (integer cents-style)
  createdAt    DateTime     @default(now())
  frozen       Boolean      @default(false)
  flags        Json         @default("{}")
  bets         Bet[]
  tickets      SupportTicket[]
}

model Game {
  id           String       @id              // e.g. "coin-flip", "roulette-v1"
  kind         String                        // "coin_flip" | "roulette" | ...
  version      Int          @default(1)
  mathConfig   Json         @default("{}")   // owned by Math agent
  deployedAt   DateTime     @default(now())
  bets         Bet[]
}

model Bet {
  id           String       @id @default(cuid())
  playerId     String
  gameId       String
  stake        BigInt
  outcome      Json                          // game-specific result payload
  payout       BigInt                        // gross payout to player (0 = lost)
  rngSeed      String                        // hex; commit-reveal capable
  settledAt    DateTime     @default(now())
  player       Player       @relation(fields: [playerId], references: [id])
  game         Game         @relation(fields: [gameId], references: [id])

  @@index([gameId, settledAt])
  @@index([playerId, settledAt])
}

enum WalletKind {
  operating
  payout_reserve
  marketing_pool
  runway
}

model TreasuryWallet {
  name         String       @id              // unique name like "ops-main"
  kind         WalletKind
  balance      BigInt       @default(0)
  movesOut     TreasuryMove[] @relation("MoveFrom")
  movesIn      TreasuryMove[] @relation("MoveTo")
}

model TreasuryMove {
  id           String       @id @default(cuid())
  fromWallet   String
  toWallet     String
  amount       BigInt
  reason       String
  byRole       String                        // "CFO" | "system" | etc.
  at           DateTime     @default(now())
  from         TreasuryWallet @relation("MoveFrom", fields: [fromWallet], references: [name])
  to           TreasuryWallet @relation("MoveTo",   fields: [toWallet],   references: [name])

  @@index([at])
}

model Campaign {
  id           String       @id @default(cuid())
  channel      String                        // "twitter_ads" | "telegram" | "affiliates" | ...
  audience     Json         @default("{}")
  budget       BigInt
  copy         String
  status       String                        // "draft" | "live" | "paused" | "complete"
  byRole       String       @default("CMO")
  createdAt    DateTime     @default(now())
  endsAt       DateTime?

  @@index([status, channel])
}

model SupportTicket {
  id              String   @id @default(cuid())
  playerId        String
  body            String
  status          String   @default("open")  // "open" | "in_progress" | "resolved"
  repliedByRole   String?
  createdAt       DateTime @default(now())
  updatedAt       DateTime @updatedAt
  player          Player   @relation(fields: [playerId], references: [id])

  @@index([status])
}

// --- Org nervous system ---

enum MessagePriority {
  low
  normal
  high
  urgent
}

model Message {
  id           String       @id @default(cuid())
  fromRole     String
  toRole       String
  subject      String
  bodyMd       String
  refsMdPaths  String[]     @default([])     // pointers to /org/* files
  threadId     String?
  repliedToId  String?
  priority     MessagePriority @default(normal)
  sentAt       DateTime     @default(now())
  readAt       DateTime?
  repliedAt    DateTime?

  @@index([toRole, readAt])
  @@index([threadId])
}

enum EventSeverity {
  low
  medium
  high
  critical
}

model Event {
  id               String       @id @default(cuid())
  type             String                          // see §7b of spec
  source           String                          // "system" | "human" | role name
  payload          Json         @default("{}")
  severity         EventSeverity @default(low)
  createdAt        DateTime     @default(now())
  processedByRole  String?
  processedAt      DateTime?

  @@index([type, createdAt])
  @@index([severity, processedAt])
}

model AgentRun {
  id                 String   @id @default(cuid())
  role               String
  reason             String                       // "cron" | "event" | "directive" | "human-memo"
  wakePacketSummary  Json     @default("{}")
  startedAt          DateTime @default(now())
  endedAt            DateTime?
  tokensIn           Int      @default(0)
  tokensOut          Int      @default(0)
```

```ts
export interface SelftestResult {

  tables: string[];
}

export async function selftest(): Promise<SelftestResult> {
  const rows = await prisma.$queryRawUnsafe<{ tablename: string }[]>(
```
